cmake_minimum_required(VERSION 3.22)
project(zapdmr_car_vad C CXX)
set(CMAKE_CXX_STANDARD 11)
file(GLOB VAD_C CONFIGURE_DEPENDS "webrtc/common_audio/signal_processing/*.c" "webrtc/common_audio/third_party/*.c" "webrtc/common_audio/vad/*.c")
add_library(carvad_core STATIC ${VAD_C} webrtc/rtc_base/checks.cc)
set_target_properties(carvad_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(carvad_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
 target_compile_definitions(carvad_core PUBLIC WEBRTC_WIN)
else()
 target_compile_definitions(carvad_core PUBLIC WEBRTC_POSIX)
endif()
if(ANDROID)
 add_library(zapcarvad SHARED car_vad_jni.cpp)
 target_link_libraries(zapcarvad carvad_core log)
 target_link_options(zapcarvad PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(carvad_check car_vad_check.cpp)
 target_link_libraries(carvad_check carvad_core)
endif()
