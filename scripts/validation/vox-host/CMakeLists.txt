cmake_minimum_required(VERSION 3.22)
project(zapdmr_host_vad_jni C CXX)
get_filename_component(ZAPDMR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT DEFINED JAVA_HOME)
 message(FATAL_ERROR "Pass -DJAVA_HOME=<JDK> to configure the host JNI test")
endif()
set(VAD_SOURCE "${ZAPDMR_ROOT}/android/app/src/main/cpp/car-vad")
add_subdirectory("${VAD_SOURCE}" vadcore)
add_library(zapcarvad SHARED "${VAD_SOURCE}/car_vad_jni.cpp")
target_include_directories(zapcarvad PRIVATE "${JAVA_HOME}/include" "${JAVA_HOME}/include/win32")
target_link_libraries(zapcarvad carvad_core)
